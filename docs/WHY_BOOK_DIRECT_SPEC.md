### Price Calculator Logic
```typescript
// lib/priceCalculator.ts
export const OTA_COMMISSIONS = {
  getyourguide: 0.30,
  klook: 0.30,
  viator: 0.25,
  airbnb: 0.20,
} as const;

export function calculateOTAPrice(basePriceIdr: number, ota: keyof typeof OTA_COMMISSIONS) {
  const commission = OTA_COMMISSIONS[ota];
  const fee = Math.round(basePriceIdr * commission);
  return {
    basePrice: basePriceIdr,
    fee,
    total: basePriceIdr + fee,
    savings: fee,
    savingsPct: Math.round((fee / (basePriceIdr + fee)) * 100),
  };
}
```
